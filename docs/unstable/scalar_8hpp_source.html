85 template <typename T>
86 requires(std::is_arithmetic_v<T> or is_complex_v<T> or std::is_same_v<T, dcplx_t>)
87 void h5_read(group g, std::string const &name, T &x) {
88 // backward compatibility to read complex values stored the old way (in a subgroup)
89 if constexpr (is_complex_v<T>) {
90 if (g.has_subgroup(name)) {
91 group gr = g.open_group(name);
92 H5_ASSERT(gr.has_key("r") and gr.has_key("i"));
93 double r = NAN, i = NAN;
94 h5_read(gr, "r", r);
95 h5_read(gr, "i", i);
96 x = std::complex<double>{r, i};
97 return;
98 }
99 }
100
101 // get dataset_info
102 auto ds_info = array_interface::get_dataset_info(g, name);
103
104 // read complex values stored as a compound HDF5 datatype
105 if constexpr (is_complex_v<T>) {
106 if (hdf5_type_equal(ds_info.ty, hdf5_type<dcplx_t>())) {
107 h5_read(g, name, reinterpret_cast<dcplx_t &>(x)); // NOLINT (reinterpret_cast is safe here)
108 return;
109 }
110 }
111
112 // read scalar value
113 array_interface::read(g, name, array_interface::array_view_from_scalar(x));
114 }
