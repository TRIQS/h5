34 /**
35 * @ingroup data_model
36 * @brief A handle to an HDF5 group.
37 *
38 * @details This class inherits from the general h5::object class. Each group stores the parent h5::file
39 * to which it belongs.
40 *
41 * It provides various methods to simplify the creation of new and opening of existing groups, subgroups,
42 * datasets and softlinks within the current group.
43 */
44 class group : public object {
45 // File to which the group belongs.
46 file parent_file;
47
48 public:
49 /// Default constructor (only necessary for the Python interface).
50 group() = default;
51
52 /**
53 * @brief Constructor to open the root ("/") group in the given file.
54 * @param f h5::file.
55 */
56 group(file f);
57
58 private:
59 // Copy the given object and parent file (only used internally to open/create a subgroup).
60 group(object obj, file _parent_file) : object{obj}, parent_file{std::move(_parent_file)} {}
61
62 public:
63 /// Get the name of the group.
64 [[nodiscard]] std::string name() const;
65
66 /// Get the parent file to which the group belongs.
67 [[nodiscard]] file get_file() const { return parent_file; }
68
69 /**
70 * @brief Check if a link with the given key exists in the group.
71 *
72 * @param key Name of the link.
73 * @return True if the link exists, false otherwise.
74 */
75 [[nodiscard]] bool has_key(std::string const &key) const;
76
77 /**
78 * @brief Check if a subgroup with the given key exists in the group and is accessible.
79 *
80 * @param key Name of the subgroup.
81 * @return True if the subgroup exists and can be accessed, false otherwise.
82 */
83 [[nodiscard]] bool has_subgroup(std::string const &key) const;
84
85 /**
86 * @brief Check if a dataset with the given key exists in the group and is accessible.
87 *
88 * @param key Name of the dataset.
89 * @return True if the dataset exists, false otherwise.
90 */
91 [[nodiscard]] bool has_dataset(std::string const &key) const;
92
93 /**
94 * @brief Remove a link with the given key from the group.
95 *
96 * @details It simply calls `H5Ldelete` to delete the link. If the given link does not exist, it throws
97 * an exception if `error_if_absent == true`, otherwise it does nothing.
98 *
99 * @param key Name of the link to be removed.
100 * @param error_if_absent If true, throws an exception if the key is not the name of a link in the group.
101 */
102 void unlink(std::string const &key, bool error_if_absent = false) const;
103
104 /**
105 * @brief Open a subgroup with the given key in the group.
106 *
107 * @details If the given key is empty, a handle to the current group is returned. Throws an exception if the
108 * subgroup fails to be opened.
109 *
110 * @param key Name of the subgroup.
111 * @return A handle to the opened subgroup.
112 */
113 [[nodiscard]] group open_group(std::string const &key) const;
114
115 /**
116 * @brief Create a subgroup with the given key in the group.
117 *
118 * @details If a subgroup with the given key already exists, it is unlinked first if `delete_if_exists == true`.
119 * If the given key is empty, a handle to the current group is returned. Throws an exception if the subgroup fails
120 * to be created.
121 *
122 * @param key Name of the subgroup to be created.
123 * @param delete_if_exists If true, unlink first an existing subgroup with the same name.
124 * @return A handle to the created subgroup.
125 */
126 group create_group(std::string const &key, bool delete_if_exists = true) const;
127
128 /**
129 * @brief Create a softlink with the given key to a target with a given target key in this group.
130 *
131 * @details Does nothing if the key or target key is empty. If `delete_if_exists == true`, it first unlinks
132 * an existing link with the same name. Throws an exception if the target does not exist, if a link with
133 * the given key already exists and `delete_if_exists == false`, or if the softlink fails to be created.
134 *
135 * @param target_key Name of target.
136 * @param key Name of the softlink to be created.
137 * @param delete_if_exists If true, unlink first an existing key with the same name.
138 */
139 void create_softlink(std::string const &target_key, std::string const &key, bool delete_if_exists = true) const;
140
141 /**
142 * @brief Open a dataset with the given key in the group.
143 *
144 * @details Throws an exception if there exists no link with the given key or if the dataset fails to be opened.
145 *
146 * @param key Name of the dataset.
147 * @return A handle to the opened dataset.
148 */
149 [[nodiscard]] dataset open_dataset(std::string const &key) const;
150
151 /**
152 * @brief Create a dataset with the given key, datatype, dataspace and dataset creation property list in this group.
153 *
154 * @details It first unlinks an existing dataset with the same name. Throws an exception if the dataset fails to be
155 * created.
156 *
157 * @param key Name of the dataset to be created.
158 * @param ty h5::datatype.
159 * @param sp h5::dataspace.
160 * @param pl Dataset creation property list.
161 * @return A handle to the created dataset.
162 */
163 dataset create_dataset(std::string const &key, datatype ty, dataspace sp, hid_t pl) const;
164
165 /**
166 * @brief Create a dataset with the given key, datatype and dataspace in this group.
167 *
168 * @details It simply calls group::create_dateset with the default dataset creation property list.
169 *
170 * @param key Name of the dataset to be created.
171 * @param ty h5::datatype.
172 * @param sp h5::dataspace.
173 * @return A handle to the created dataset.
174 */
175 dataset create_dataset(std::string const &key, datatype ty, dataspace sp) const;
176
177 /**
178 * @brief Get all the names of the subgroups in the current group.
179 * @return A vector with the names of all the subgroups.
180 */
181 [[nodiscard]] std::vector<std::string> get_all_subgroup_names() const;
182
183 /**
184 * @brief Get all the names of the datasets in the current group.
185 * @return A vector with the names of all the datasets.
186 */
187 [[nodiscard]] std::vector<std::string> get_all_dataset_names() const;
188
189 /**
190 * @brief Get all the names of the subgroups and datasets in the current group.
191 * @return A vector with the names of all the subgroups and datasets.
192 */
193 [[nodiscard]] std::vector<std::string> get_all_subgroup_dataset_names() const;
194 };
