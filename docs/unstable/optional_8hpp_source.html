45 /**
46 * @brief Write a std::optional to an HDF5 dataset/subgroup (if it is set).
47 *
48 * @details Calls the specialized `h5_write` function for the value type of the std::optional.
49 *
50 * @tparam T Value type of std::optional.
51 * @param g h5::group in which the dataset/subgroup is created.
52 * @param name Name of the dataset/subgroup to which the std::optional value is written.
53 * @param opt std::optional to be written.
54 */
55 template <typename T>
56 void h5_write(group g, std::string const &name, std::optional<T> const &opt) {
57 if (opt) h5_write(g, name, *opt);
58 }
59
60 /**
61 * @brief Read a std::optional from an HDF5 dataset/subgroup.
62 *
63 * @details Calls the specialized `h5_read` function for the value type of the std::optional.
64 *
65 * @tparam T Value type of std::optional.
66 * @param g h5::group containing the dataset/subgroup.
67 * @param name Name of the dataset/subgroup from which the std::optional value is read.
68 * @param opt std::optional to read into.
69 */
70 template <typename T>
71 void h5_read(group g, std::string name, std::optional<T> &opt) {
72 opt.reset();
73 if (g.has_key(name)) opt.emplace(h5_read<T>(g, name));
74 }
