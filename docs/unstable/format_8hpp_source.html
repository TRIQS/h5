17/**
18 * @file
19 * @brief Provides utilities for reading and writing `hdf5_format` tags.
20 * @details An `hdf5_format` tag is a string that describes the type of object stored in an HDF5 file. A type `T` which is
21 * HDF5 readable/writeable should do one of the following:
22 * - implement a static member function `static hdf5_format() -> std::string` or
23 * - specialize the `hdf5_format_impl` struct for `T` and provide an implementation for the member `static invoke() -> std::string`.
24 */

43 /**
44 * @brief Default type trait to get the `hdf5_format` tag of type `T` by calling its static member function
45 * `T::hdf5_format()`.
46 *
47 * @tparam T Type for which the `hdf5_format` tag is to be retrieved.
48 */
49 template <typename T>
50 struct hdf5_format_impl {
51 static std::string invoke() { return T::hdf5_format(); }
52 };
53
54#define H5_SPECIALIZE_FORMAT2(X, Y)
55 /** @brief Specialization of h5::hdf5_format_impl for X. */
56 template <>
57 struct hdf5_format_impl<X> {
58 static std::string invoke() { return H5_AS_STRING(Y); }
59 }

75 /**
76 * @brief Get the `hdf5_format` tag of type `T`.
77 *
78 * @tparam T Type for which the `hdf5_format` tag is to be retrieved..
79 * @return std::string containing the `hdf5_format` tag.
80 */
81 template <typename T>
82 std::string get_hdf5_format() {
83 return hdf5_format_impl<T>::invoke();
84 }

86 /**

92 template <typename T>
93 std::string get_hdf5_format(T const &) {
94 return hdf5_format_impl<T>::invoke();

157 * @details Throws a std::runtime_error if the tags don't match.
158 *
159 * @tparam T Type for which the `hdf5_format` tag is to be checked.
160 * @param g h5::group to be checked.
161 * @param ignore_if_absent If true, the assertion is ignored if the group does not have a 'Format' attribute.
