34 /**
35 * @ingroup data_model
36 * @brief A handle to an HDF5 file.
37 *
38 * @details This class inherits from the general h5::object class. It simply wraps the HDF5 functions
39 * to open an existing file or to create a new file.
40 *
41 * An h5::file is automatically closed when it goes out of scope, i.e. its reference count is decreased.
42 */
43 class file : public object {
44 public:
45 /**
46 * @brief Default constructor creates a buffered memory file.
47 * @details It modifies the file access property list to use the `H5FD_CORE` driver. It can be used for
48 * serializing and deserializing data (see h5::serialize and h5::deserialize).
49 */
50 file();
51
52 /**
53 * @brief Constructor to open an existing file or to create a new file on disk.
54 *
55 * @details The file is opened in the specified mode. The following modes are available:
56 *
57 * - 'r': Open an existing file in read only mode (calls `H5Fopen` with `H5F_ACC_RDONLY`).
58 * - 'w': Create a new file or overwrite an existing file in read-write mode (calls `H5Fcreate` with `H5F_ACC_TRUNC`).
59 * - 'a': Create a new file or append to an existing file in read-write mode (calls `H5Fcreate` with `H5F_ACC_EXCL`
60 * or `H5Fopen` with `H5F_ACC_RDWR` in case the file already exists).
61 * - 'e': Create a new file if the file does not already exists, otherwise throw an exception (calls `H5Fcreate` with
62 * `H5F_ACC_EXCL`)
63 *
64 * @param name Name of the file.
65 * @param mode Mode in which to open the file.
66 */
67 file(const char *name, char mode);
68
69 /**
70 * @brief Constructor to open an existing file or to create a new file on disk.
71 * @details See file::file(const char*, char) for a more detailed description.
72 */
73 file(std::string const &name, char mode) : file(name.c_str(), mode) {}
74
75 /// Get the name of the file.
76 [[nodiscard]] std::string name() const;
77
78 /// Flush the file by calling `H5Fflush`.
79 void flush();
80
81 private:
82 // Constructor to create a buffered memory file with an initial file image of a given size.
83 file(const std::byte *buf, size_t size);
84
85 public:
86 /**
87 * @brief Constructor to create a buffered memory file from a byte buffer.
88 * @param buf Byte buffer.
89 */
90 file(std::span<std::byte> const &buf) : file(buf.data(), buf.size()) {}
91
92 /**
93 * @brief Constructor to create a buffered memory file from a byte buffer.
94 * @param buf Byte buffer.
95 */
96 file(std::vector<std::byte> const &buf) : file(buf.data(), buf.size()) {}
97
98 /// Get a copy of the associated byte buffer.
99 [[nodiscard]] std::vector<std::byte> as_buffer() const;
100 };
