149 /**
150 * @brief Stores an arbitrary number of strings in a 1-dimensional std::vector<char>.
151 *
152 * @details Each string is assumed to have the same length. If a string is shorter than this length, it is padded
153 * with zeros. The `lengths` member should have the following entries: the number of strings in each dimension and
154 * the max. allowed length of the strings + 1. For example, if the original strings are stored in a 2-dimensional
155 * array of size `MxN` and the longest string is of length `L`, then `lengths = {M, N, L+1}`.
156 *
157 * The HDF5 datatype is a fixed-length string of size `lengths.back()` and the HDF5 dataspace is an n-dimensional
158 * array of fixed-sized strings.
159 */
160 struct char_buf {
161 /// Stores strings in a 1-dimensional vector.
162 std::vector<char> buffer;
163
164 /// Stores the number of strings in each dimension and the max. allowed length of the strings + 1.
165 v_t lengths;
166
167 /// Get the HDF5 datatype.
168 [[nodiscard]] datatype dtype() const;
169
170 /// Get the HDF5 dataspace.
171 [[nodiscard]] dataspace dspace() const;
172 };
