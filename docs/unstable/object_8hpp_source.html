36 /**
37 * @brief A generic handle for HDF5 objects.
38 *
39 * @details It simply stores the h5::hid_t of the corresponding HDF5 object.
40 *
41 * More specific HDF5 objects, like h5::file or h5::group, inherit from this class. Since it lacks a virtual
42 * destructor, the derived classes should not be deleted through a pointer to this class. It is recommended
43 * to use the derived classes whenever possible.
44 *
45 * HDF5's reference counting system is similar to Python's. This class handles the proper reference counting
46 * using a RAII pattern (hence exception safe). Depending on how the object is constructed, it either
47 * increases the reference count associated with the HDF5 object or steals it.
48 */
49 class object {
50 protected:
51 // Wrapped HDF5 ID.
52 hid_t id = 0; // NOLINT (protected member is wanted here)
53
54 public:
55 /**
56 * @brief Create an h5::object for a given HDF5 ID and increase its reference count.
57 *
58 * @param id HDF5 ID.
59 * @return h5::object.
60 */
61 [[nodiscard]] static object from_borrowed(hid_t id);
62
63 /**
64 * @brief Construct a new h5::object for a given HDF5 ID by taking ownership, i.e. without increasing the
65 * reference count.
66 *
67 * @details If no ID is given, it is set to zero (default).
68 *
69 * @param id HDF5 ID.
70 */
71 object(hid_t id = 0) : id(id) {}
72
73 /**
74 * @brief Copy constructor copies the underlying HDF5 ID and increases its reference count.
75 * @param x Object to copy.
76 */
77 object(object const &x);
78
79 /**
80 * @brief Move constructor steals the underlying HDF5 ID without increasing its reference count.
81 * @param x Object to move.
82 */
83 object(object &&x) noexcept : id(x.id) { x.id = 0; }
84
85 /**
86 * @brief Copy assignment operator copies the underlying HDF5 ID and increases its reference count.
87 * @param x Object to copy.
88 */
89 object &operator=(object const &x) {
90 operator=(object(x));
91 return *this;
92 }
93
94 /**
95 * @brief Move assignment operator steals the underlying HDF5 ID without increasing its reference count.
96 * @param x Object to move.
97 */
98 object &operator=(object &&x) noexcept;
99
100 /// Destructor decreases the reference count and sets the object's ID to zero.
101 ~object() { close(); }
102
103 /// Release the HDF5 handle by decreasing the reference count and by setting the object's ID to zero.
104 void close();
105
106 /// User-defined conversion to h5::hid_t.
107 operator hid_t() const { return id; }
108
109 /// Get the current reference count.
110 [[nodiscard]] int get_ref_count() const;
111
112 /// Ensure that the wrapped HDF5 ID is valid (by calling `H5Iis_valid`).
113 [[nodiscard]] bool is_valid() const;
114 };

143 // Map a C++ type to an HDF5 type (specializations are in object.cpp).
144 template <typename T>
145 hid_t hid_t_of();
146
147 } // namespace detail

149 /**
150 * @brief Map a given C++ type to an HDF5 datatype.
151 *
152 * @tparam T C++ type.
153 * @return h5::datatype object corresponding to the given C++ type.
154 */
155 template <typename T>
156 [[nodiscard]] datatype hdf5_type() {
157 return object::from_borrowed(detail::hid_t_of<T>());
158 }
159
160 /**
161 * @brief Get the name of an h5::datatype (for error messages).
162 *
163 * @details Throws an exception if the datatype is not supported.
164 *
165 * @param dt h5::datatype.
166 * @return String representation of the datatype.
167 */
168 [[nodiscard]] std::string get_name_of_h5_type(datatype dt);
169
170 /**
171 * @brief Get the HDF5 type stored in a given h5::dataset.
172 *
173 * @param ds h5::dataset.
174 * @return h5::datatype of the given h5::dataset.
175 */
176 [[nodiscard]] datatype get_hdf5_type(dataset ds);
177
178 /**
179 * @brief Check if two HDF5 datatypes are equal.
180 *
181 * @details For string types, this function only checks if they are both of the class `H5T_STRING`.
182 * It does not take into account the size, character set, etc.
183 *
184 * Otherwise, it simply uses `H5Tequal`. Throws an exception if the HDF5 call fails.
185 *
186 * @param dt1 h5::datatype #1.
187 * @param dt2 h5::datatype #2.
188 * @return True, if the two datatypes are equal.
189 */
190 [[nodiscard]] bool hdf5_type_equal(datatype dt1, datatype dt2);
