39 /**
40 * @brief Generic implementation for reading from an HDF5 dataset/subgroup.
41 *
42 * @details It calls the static member function `T::h5_read_construct(group, T &)` in case `T` is not default constructible,
43 * otherwise it calls the specialized `h5_read(group, std::string const &, T &)` with a default constructed `T` object.
44 *
45 * @tparam T C++ type to be read.
46 * @param g h5::group containing the dataset/subgroup from which to read from.
47 * @param key Name of the dataset/subgroup.
48 * @return Data read from the dataset/subgroup.
49 */
50 template <typename T>
51 T h5_read(group g, std::string const &key) {
52 // default constructible
53 if constexpr (std::is_default_constructible_v<T>) {
54 T x{};
55 h5_read(g, key, x);
56 return x;
57 } else { // not default constructible
58 return T::h5_read_construct(g, key);
59 }
60 }

190 /**
191 * @brief Check if an HDF5 dataset/subgroup with the given key exists in the given parent group before performing the read.
192 *
193 * @details If the key exists, it simply calls the specialized `h5_read(group, std::string const &, T &)` for the given `T`,
194 * otherwise it does nothing.
195 *
196 * @tparam T C++ type to be read.
197 * @param g h5::group containing the dataset/subgroup from which to read from.
198 * @param key Name of the dataset/subgroup.
199 * @param x Variable to read into.
200 * @return True, if the dataset/subgroup exists and reading was successful, false otherwise.
201 */
202 template <typename T>
203 inline bool try_read(group g, std::string const &key, T &x) {
204 if (g.has_key(key)) {
205 h5_read(g, key, x);
206 return true;
207 }
208 return false;
209 }
